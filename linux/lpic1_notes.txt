# Read files
cat - concatenate files and print on the standard output
cat filename.txt

# More commands for reading file content.
more - file perusal filter for crt viewing
more /var/log/auth.log

less - opposite of more
less /var/log/auth.log
# moze pretrazivati sa /pojam

head - output the first part of files
head -n 5 filename

tail - output the last part of filesl
tail -n 10 -f /var/log/auth.log )Moze pratiti u zivo promjene sa -f)
